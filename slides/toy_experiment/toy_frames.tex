\begin{frame}{Toy Experiment: Results}
  \small
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{}p{0.2\textwidth}p{0.36\textwidth}p{0.36\textwidth}@{}}
    & \textbf{Example 1 (clean)} & \textbf{Example 2 (noisy)} \\
    \arrayrulecolor{piirule}\midrule
    \piiTag{Private?} &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace no real value sent &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace no real value sent \\
    \piiTag{Enough context?} &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace all 3 placeholders used &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace all 3 placeholders used \\
    \piiTag{Usable answer?} &
      \textbf{No}\enspace sign-off left as \texttt{<NAME\_1} &
      \textbf{No}\enspace same, and the typos come back (``41 maple stret'') \\
    \arrayrulecolor{piirule}\bottomrule
  \end{tabular}

  \vspace{0.6em}
  \begin{itemize}\setlength{\itemsep}{0.35em}
    \item[\textcolor{piiaccent}{\textbullet}] \textbf{The loop works:} the LLM wrote a usable message
      without seeing any real value.
    \item[\textcolor{piiaccent}{\textbullet}] \textbf{Unmasking is the weak step:} the LLM dropped the
      closing ``>'' of its last placeholder, which simple replacement cannot restore.
    \item[\textcolor{piiaccent}{\textbullet}] \textbf{RQ2 hint:} typing the name in lowercase cut its
      confidence from 0.999 to 0.686; nothing leaked, but close to the cut-off.
  \end{itemize}
\end{frame}
