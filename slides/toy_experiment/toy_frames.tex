\begin{frame}{Toy Experiment: Results}
  \small
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{}p{0.2\textwidth}p{0.36\textwidth}p{0.36\textwidth}@{}}
    & \textbf{Example 1 (clean)} & \textbf{Example 2 (noisy)} \\
    \arrayrulecolor{piirule}\midrule
    \piiTag{Private?} &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace no real value sent &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace no real value sent \\
    \piiTag{Enough context?} &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace all 3 placeholders used &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace all 3 placeholders used \\
    \piiTag{Usable answer?} &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace all 3 values restored &
      \textcolor{piiaccent}{\textbf{Yes}}\enspace all 3 values restored \\
    \arrayrulecolor{piirule}\bottomrule
  \end{tabular}

  \vspace{0.6em}
  \begin{itemize}\setlength{\itemsep}{0.35em}
    \item[\textcolor{piiaccent}{\textbullet}] \textbf{The loop works:} GPT-5.5 wrote the message without
      seeing any real value, and every placeholder was restored. Same result in 3 runs.
    \item[\textcolor{piiaccent}{\textbullet}] \textbf{RQ2 hint:} typing the name in lowercase cut its
      confidence from 0.999 to 0.686; nothing leaked, but close to the cut-off.
    \item[\textcolor{piiaccent}{\textbullet}] \textbf{The noise comes back:} unmasking restores what the
      user typed, so the noisy answer reads ``this is minji park from 41 maple stret''.
  \end{itemize}
\end{frame}
